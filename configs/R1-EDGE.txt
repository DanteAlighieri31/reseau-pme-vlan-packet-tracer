! =====================================================================
!  R1-EDGE - Routeur de bordure (Cisco 2911)
!  Role : passerelle Internet (NAT/PAT), serveur DHCP, OSPF, filtrage WAN
!  Projet : Reseau segmente de la PME Arvex - Cisco Packet Tracer
! =====================================================================
!  Coller bloc par bloc dans l'onglet CLI de l'equipement
!  (repondre "no" a la question "initial configuration dialog")
! =====================================================================
enable
configure terminal
!
! ---------------------------------------------------------------------
! 1. Parametres generaux
! ---------------------------------------------------------------------
hostname R1-EDGE
no ip domain-lookup
ip domain-name arvex.lan
service password-encryption
service timestamps log datetime msec
enable secret Arvex-Enable-2026
username admin privilege 15 secret Arvex-Admin-2026
banner motd #
*****************************************************************
  R1-EDGE - Acces reserve au personnel autorise.
  Toute connexion est journalisee.
*****************************************************************
#
!
! ---------------------------------------------------------------------
! 2. Interfaces
! ---------------------------------------------------------------------
interface GigabitEthernet0/0
 description LAN - Lien route vers SW-CORE Gi0/1
 ip address 10.10.255.1 255.255.255.252
 ip nat inside
 no shutdown
 exit
!
interface GigabitEthernet0/1
 description WAN - Lien vers le FAI (ISP Gi0/0)
 ip address 203.0.113.2 255.255.255.252
 ip nat outside
 no cdp enable
 no shutdown
 exit
!
interface GigabitEthernet0/2
 description NON UTILISE
 shutdown
 exit
!
! ---------------------------------------------------------------------
! 3. Routage : route par defaut vers le FAI, diffusee en OSPF
! ---------------------------------------------------------------------
ip route 0.0.0.0 0.0.0.0 203.0.113.1
!
router ospf 1
 router-id 1.1.1.1
 network 10.10.255.0 0.0.0.3 area 0
 default-information originate
 exit
!
! ---------------------------------------------------------------------
! 4. NAT/PAT : tous les reseaux internes sortent avec l'IP publique
! ---------------------------------------------------------------------
access-list 1 remark Reseaux internes autorises a sortir sur Internet
access-list 1 permit 10.10.0.0 0.0.255.255
ip nat inside source list 1 interface GigabitEthernet0/1 overload
!
! ---------------------------------------------------------------------
! 5. Serveur DHCP (les requetes sont relayees par SW-CORE)
! ---------------------------------------------------------------------
ip dhcp excluded-address 10.10.10.1 10.10.10.20
ip dhcp excluded-address 10.10.20.1 10.10.20.20
ip dhcp excluded-address 10.10.30.1 10.10.30.20
ip dhcp excluded-address 10.10.50.1 10.10.50.20
!
ip dhcp pool VLAN10-DIRECTION
 network 10.10.10.0 255.255.255.0
 default-router 10.10.10.1
 dns-server 10.10.40.10
 exit
ip dhcp pool VLAN20-COMPTA
 network 10.10.20.0 255.255.255.0
 default-router 10.10.20.1
 dns-server 10.10.40.10
 exit
ip dhcp pool VLAN30-COMMERCIAL
 network 10.10.30.0 255.255.255.0
 default-router 10.10.30.1
 dns-server 10.10.40.10
 exit
ip dhcp pool VLAN50-INFORMATIQUE
 network 10.10.50.0 255.255.255.0
 default-router 10.10.50.1
 dns-server 10.10.40.10
 exit
!
! ---------------------------------------------------------------------
! 6. Filtrage WAN : anti-usurpation (RFC 1918 / boucle locale)
! ---------------------------------------------------------------------
ip access-list extended ACL-WAN-IN
 remark Refuser les adresses privees qui arrivent depuis Internet
 deny ip 10.0.0.0 0.255.255.255 any
 deny ip 172.16.0.0 0.15.255.255 any
 deny ip 192.168.0.0 0.0.255.255 any
 deny ip 127.0.0.0 0.255.255.255 any
 permit ip any any
 exit
!
interface GigabitEthernet0/1
 ip access-group ACL-WAN-IN in
 exit
!
! ---------------------------------------------------------------------
! 7. Journalisation centralisee et synchronisation de l'heure
! ---------------------------------------------------------------------
logging host 10.10.40.10
ntp server 10.10.40.10
!
! ---------------------------------------------------------------------
! 8. Administration : SSH uniquement, depuis le VLAN 50 INFORMATIQUE
! ---------------------------------------------------------------------
access-list 10 remark Postes autorises a administrer le reseau
access-list 10 permit 10.10.50.0 0.0.0.255
!
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
!
line console 0
 login local
 exec-timeout 5 0
 logging synchronous
 exit
!
line vty 0 4
 login local
 transport input ssh
 access-class 10 in
 exec-timeout 10 0
 exit
!
end
write memory
