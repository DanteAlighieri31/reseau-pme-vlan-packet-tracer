! =====================================================================
!  SW-CORE - Switch coeur de niveau 3 (Cisco 3560-24PS)
!  Role : routage inter-VLAN, relais DHCP, ACL, racine Spanning Tree, OSPF
!  Projet : Reseau segmente de la PME Arvex - Cisco Packet Tracer
! =====================================================================
!  Coller bloc par bloc dans l'onglet CLI de l'equipement
! =====================================================================
enable
configure terminal
!
! ---------------------------------------------------------------------
! 1. Parametres generaux
! ---------------------------------------------------------------------
hostname SW-CORE
no ip domain-lookup
ip domain-name arvex.lan
service password-encryption
service timestamps log datetime msec
enable secret Arvex-Enable-2026
username admin privilege 15 secret Arvex-Admin-2026
banner motd #
*****************************************************************
  SW-CORE - Acces reserve au personnel autorise.
  Toute connexion est journalisee.
*****************************************************************
#
!
! ---------------------------------------------------------------------
! 2. VLANs (VTP transparent : chaque switch gere sa propre base VLAN)
! ---------------------------------------------------------------------
vtp mode transparent
!
vlan 10
 name DIRECTION
vlan 20
 name COMPTA
vlan 30
 name COMMERCIAL
vlan 40
 name SERVEURS
vlan 50
 name INFORMATIQUE
vlan 99
 name ADMIN
vlan 998
 name PARKING
vlan 999
 name NATIVE
 exit
!
! ---------------------------------------------------------------------
! 3. Spanning Tree : Rapid PVST+, SW-CORE est le pont racine
! ---------------------------------------------------------------------
spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40,50,99 priority 4096
!
! ---------------------------------------------------------------------
! 4. Trunks 802.1Q vers les switches d'acces
! ---------------------------------------------------------------------
interface FastEthernet0/1
 description TRUNK vers SW-ACC1 Gi0/1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,99,999
 switchport nonegotiate
 no shutdown
 exit
!
interface FastEthernet0/2
 description TRUNK vers SW-ACC2 Gi0/1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 20,30,50,99,999
 switchport nonegotiate
 no shutdown
 exit
!
! ---------------------------------------------------------------------
! 5. Port d'acces du serveur (VLAN 40)
! ---------------------------------------------------------------------
interface FastEthernet0/10
 description SRV-INTRA
 switchport mode access
 switchport access vlan 40
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 spanning-tree portfast
 spanning-tree bpduguard enable
 no shutdown
 exit
!
! ---------------------------------------------------------------------
! 6. Ports inutilises : VLAN parking + desactives
! ---------------------------------------------------------------------
interface range FastEthernet0/3 - 9 , FastEthernet0/11 - 24 , GigabitEthernet0/2
 description NON UTILISE
 switchport mode access
 switchport access vlan 998
 shutdown
 exit
!
interface Vlan1
 shutdown
 exit
!
! ---------------------------------------------------------------------
! 7. Routage : lien route vers R1-EDGE et passerelles des VLANs (SVI)
! ---------------------------------------------------------------------
ip routing
!
interface GigabitEthernet0/1
 description Lien route vers R1-EDGE Gi0/0
 no switchport
 ip address 10.10.255.2 255.255.255.252
 no shutdown
 exit
!
interface Vlan10
 description Passerelle DIRECTION
 ip address 10.10.10.1 255.255.255.0
 ip helper-address 10.10.255.1
 no shutdown
 exit
interface Vlan20
 description Passerelle COMPTA
 ip address 10.10.20.1 255.255.255.0
 ip helper-address 10.10.255.1
 no shutdown
 exit
interface Vlan30
 description Passerelle COMMERCIAL
 ip address 10.10.30.1 255.255.255.0
 ip helper-address 10.10.255.1
 no shutdown
 exit
interface Vlan40
 description Passerelle SERVEURS
 ip address 10.10.40.1 255.255.255.0
 no shutdown
 exit
interface Vlan50
 description Passerelle INFORMATIQUE
 ip address 10.10.50.1 255.255.255.0
 ip helper-address 10.10.255.1
 no shutdown
 exit
interface Vlan99
 description Passerelle ADMIN (management)
 ip address 10.10.99.1 255.255.255.0
 no shutdown
 exit
!
router ospf 1
 router-id 2.2.2.2
 network 10.10.10.0 0.0.0.255 area 0
 network 10.10.20.0 0.0.0.255 area 0
 network 10.10.30.0 0.0.0.255 area 0
 network 10.10.40.0 0.0.0.255 area 0
 network 10.10.50.0 0.0.0.255 area 0
 network 10.10.99.0 0.0.0.255 area 0
 network 10.10.255.0 0.0.0.3 area 0
 passive-interface Vlan10
 passive-interface Vlan20
 passive-interface Vlan30
 passive-interface Vlan40
 passive-interface Vlan50
 passive-interface Vlan99
 exit
!
! ---------------------------------------------------------------------
! 8. ACL de cloisonnement (appliquees en entree des SVI)
!    - Personne sauf INFORMATIQUE n'accede au VLAN 99 ADMIN
!    - COMMERCIAL n'accede pas a la COMPTA
!    - Anti-usurpation : chaque VLAN n'accepte que ses propres adresses
!    - "udp eq 67" laisse passer les demandes DHCP (adresse source 0.0.0.0)
! ---------------------------------------------------------------------
ip access-list extended ACL-DIRECTION-IN
 remark Requetes DHCP
 permit udp any any eq 67
 remark Interdire le reseau d administration
 deny ip any 10.10.99.0 0.0.0.255
 remark Anti-usurpation
 permit ip 10.10.10.0 0.0.0.255 any
 deny ip any any
 exit
!
ip access-list extended ACL-COMPTA-IN
 remark Requetes DHCP
 permit udp any any eq 67
 remark Interdire le reseau d administration
 deny ip any 10.10.99.0 0.0.0.255
 remark Anti-usurpation
 permit ip 10.10.20.0 0.0.0.255 any
 deny ip any any
 exit
!
ip access-list extended ACL-COMMERCIAL-IN
 remark Requetes DHCP
 permit udp any any eq 67
 remark Interdire le reseau d administration
 deny ip any 10.10.99.0 0.0.0.255
 remark Interdire la comptabilite
 deny ip any 10.10.20.0 0.0.0.255
 remark Anti-usurpation
 permit ip 10.10.30.0 0.0.0.255 any
 deny ip any any
 exit
!
ip access-list extended ACL-INFORMATIQUE-IN
 remark Requetes DHCP
 permit udp any any eq 67
 remark Anti-usurpation (acces complet pour le service informatique)
 permit ip 10.10.50.0 0.0.0.255 any
 deny ip any any
 exit
!
interface Vlan10
 ip access-group ACL-DIRECTION-IN in
 exit
interface Vlan20
 ip access-group ACL-COMPTA-IN in
 exit
interface Vlan30
 ip access-group ACL-COMMERCIAL-IN in
 exit
interface Vlan50
 ip access-group ACL-INFORMATIQUE-IN in
 exit
!
! ---------------------------------------------------------------------
! 9. Journalisation centralisee et synchronisation de l'heure
! ---------------------------------------------------------------------
logging host 10.10.40.10
ntp server 10.10.40.10
!
! ---------------------------------------------------------------------
! 10. Administration : SSH uniquement, depuis le VLAN 50 INFORMATIQUE
! ---------------------------------------------------------------------
access-list 10 remark Postes autorises a administrer le reseau
access-list 10 permit 10.10.50.0 0.0.0.255
!
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
!
line console 0
 login local
 exec-timeout 5 0
 logging synchronous
 exit
!
line vty 0 15
 login local
 transport input ssh
 access-class 10 in
 exec-timeout 10 0
 exit
!
end
write memory
